    <script>
        // Configuración de la Ruleta PRINCIPAL (45 grados por sector)
        const SECTORS_PRINCIPAL = [
            { id: 'x50', multiplier: 50.0, probability: 0.05, stopAngle: 0 },
            { id: 'x25', multiplier: 25.0, probability: 0.05, stopAngle: 45 },
            { id: 'x10', multiplier: 10.0, probability: 0.10, stopAngle: 90 },
            { id: 'x5', multiplier: 5.0, probability: 0.10, stopAngle: 135 },
            { id: 'x3', multiplier: 3.0, probability: 0.15, stopAngle: 180 },
            { id: 'x2', multiplier: 2.0, probability: 0.15, stopAngle: 225 },
            { id: 'x0', multiplier: 0.0, probability: 0.20, stopAngle: 270 },
            // SECTOR ESTRELLA BONUS - Lleva a la ruleta de bonus
            { id: 'star', multiplier: 'BONUS', probability: 0.20, stopAngle: 315, isBonus: true },
        ];

        // Configuración de la Ruleta BONUS (valores enteros más altos)
        const SECTORS_BONUS = [
            { id: 'x500', multiplier: 500, probability: 0.05, stopAngle: 0 },
            { id: 'x200', multiplier: 200, probability: 0.10, stopAngle: 45 },
            { id: 'x150', multiplier: 150, probability: 0.10, stopAngle: 90 },
            { id: 'x100', multiplier: 100, probability: 0.15, stopAngle: 135 },
            { id: 'x80', multiplier: 80, probability: 0.15, stopAngle: 180 },
            { id: 'x50', multiplier: 50, probability: 0.20, stopAngle: 225 },
            { id: 'x30', multiplier: 30, probability: 0.20, stopAngle: 270 },
            { id: 'x20', multiplier: 20, probability: 0.05, stopAngle: 315 },
        ];

        // Estado del Juego
        let balance = 1000;
        let isSpinning = false;
        let isBonusRound = false;
        let isBonusUsed = false;
        let currentBet = 0;
        let currentSectors = SECTORS_PRINCIPAL;

        // Elementos del DOM
        const balanceDisplay = document.getElementById('balance-display');
        const betInput = document.getElementById('bet-input');
        const btnSpin = document.getElementById('btn-spin');
        const wheelSVGNormal = document.getElementById('wheel-svg-normal');
        const wheelSVGBonus = document.getElementById('wheel-svg-bonus');
        const statusMessage = document.getElementById('status-message');
        const resultMessage = document.getElementById('result-message');
        const logContainer = document.getElementById('log-container');
        const gameTitle = document.getElementById('game-title');
        const bonusIndicator = document.getElementById('bonus-indicator');

        // Obtener la ruleta activa actual
        function getCurrentWheel() {
            return isBonusRound ? wheelSVGBonus : wheelSVGNormal;
        }

        // Función de utilidad para formato de moneda
        function formatCurrency(amount) {
            return amount.toLocaleString('es-ES', {
                style: 'currency',
                currency: 'EUR',
                minimumFractionDigits: 0,
                maximumFractionDigits: 0
            });
        }

        // Función para registrar eventos en el historial
        function log(message, type = 'info') {
            const entry = document.createElement('div');
            entry.className = 'log-entry py-1 ' + 
                (type === 'win' ? 'text-green-300' : 
                 type === 'bonus' ? 'text-yellow-300' : 
                 type === 'loss' ? 'text-red-300' : 'text-gray-400');
            entry.textContent = `[${new Date().toLocaleTimeString()}] ${message}`;
            logContainer.prepend(entry);
            if (logContainer.children.length > 20) {
                logContainer.lastChild.remove();
            }
        }

        // Cambiar a ruleta de bonus
        function activateBonusRound() {
            isBonusRound = true;
            isBonusUsed = false;
            currentSectors = SECTORS_BONUS;
            
            // Actualizar UI para modo bonus
            gameTitle.textContent = "🎰 RULETA BONUS 🎰";
            gameTitle.className = "text-3xl font-bold text-center mb-6 gold-gradient-text";
            bonusIndicator.classList.remove('hidden');
            btnSpin.classList.add('bonus-active');
            
            // Cambiar a ruleta bonus
            wheelSVGNormal.style.display = 'none';
            wheelSVGBonus.style.display = 'block';
            wheelSVGBonus.style.transform = 'rotate(0deg)';
            
            statusMessage.textContent = "¡MODO BONUS! Gira para multiplicadores ALTOS";
            statusMessage.className = "text-center font-semibold text-lg mb-4 h-6 gold-text";
            
            updateUI();
        }

        // Regresar a ruleta normal
        function deactivateBonusRound() {
            isBonusRound = false;
            isBonusUsed = false;
            currentSectors = SECTORS_PRINCIPAL;
            
            // Restaurar UI normal
            gameTitle.textContent = "Ruleta de la Suerte";
            gameTitle.className = "text-3xl font-bold text-center mb-6 text-indigo-400";
            bonusIndicator.classList.add('hidden');
            btnSpin.classList.remove('bonus-active');
            statusMessage.className = "text-center font-semibold text-lg mb-4 h-6";
            
            // Restaurar ruleta normal
            wheelSVGBonus.style.display = 'none';
            wheelSVGNormal.style.display = 'block';
            wheelSVGNormal.style.transform = 'rotate(0deg)';
            
            updateUI();
        }

        // Seleccionar sector ganador
        function selectWinningSector(sectors) {
            let totalWeight = sectors.reduce((sum, sector) => sum + sector.probability, 0);
            let randomNum = Math.random() * totalWeight;
            
            let cumulativeWeight = 0;
            for (const sector of sectors) {
                cumulativeWeight += sector.probability;
                if (randomNum < cumulativeWeight) {
                    return sector;
                }
            }
            return sectors[sectors.length - 1];
        }

        // Calcular ángulo de parada
        function calculateStopAngle(winningSector) {
            const targetAngle = winningSector.stopAngle;
            const deviation = (Math.random() * 36) - 18;
            let rotationOffset = 360 - ((targetAngle + deviation) % 360);
            const fullSpins = 5;
            const fullRotation = fullSpins * 360;
            return fullRotation + rotationOffset;
        }

        // Actualizar UI
        function updateUI() {
            balanceDisplay.textContent = formatCurrency(balance);
            
            let shouldDisable = isSpinning || 
                               balance < Number(betInput.value) || 
                               Number(betInput.value) <= 0 ||
                               (isBonusRound && isBonusUsed);
            
            btnSpin.disabled = shouldDisable;
            
            if (isBonusRound) {
                btnSpin.textContent = isBonusUsed ? "Bonus Usado" : "Girar (BONUS)";
            } else {
                btnSpin.textContent = "Girar";
            }
        }

        // Girar la ruleta
        function spinWheel() {
            const bet = Number(betInput.value);

            if (isBonusRound && isBonusUsed) {
                return;
            }

            if (isSpinning || bet <= 0 || balance < bet) return;

            if (isBonusRound) {
                isBonusUsed = true;
            }

            currentBet = bet;
            
            if (!isBonusRound) {
                balance -= bet;
            }
            
            isSpinning = true;
            updateUI();
            
            statusMessage.textContent = isBonusRound ? "¡Girando Ruleta BONUS!" : "¡Girando la ruleta!";
            resultMessage.textContent = "";

            const winningSector = selectWinningSector(currentSectors);
            const degrees = calculateStopAngle(winningSector);
            
            const duration = 6000;
            const currentWheel = getCurrentWheel();
            
            currentWheel.style.transition = `transform ${duration / 1000}s cubic-bezier(0.35, 0.01, 0, 1)`;
            currentWheel.style.transform = `rotate(${degrees}deg)`;

            setTimeout(() => {
                isSpinning = false;
                
                currentWheel.style.transition = 'none';
                let finalRotation = degrees % 360;
                if (finalRotation < 0) finalRotation += 360;
                currentWheel.style.transform = `rotate(${finalRotation}deg)`;

                if (winningSector.isBonus) {
                    resultMessage.innerHTML = `<span class="gold-gradient-text">🎉 ¡BONUS! Pasas a la ruleta de multiplicadores altos 🎉</span>`;
                    statusMessage.textContent = "¡MODO BONUS ACTIVADO!";
                    statusMessage.className = "text-center font-semibold text-lg mb-4 h-6 gold-text";
                    log(`⭐ CAÍSTE EN LA ESTRELLA! Bonus activado. Apuesta: ${formatCurrency(currentBet)}`, 'bonus');
                    activateBonusRound();
                    return;
                }

                const multiplier = winningSector.multiplier;
                const winAmount = currentBet * multiplier;

                let message;
                let logType;

                if (multiplier > 0) {
                    balance += winAmount;
                    
                    if (isBonusRound) {
                        message = `🎊 ¡BONUS GANADOR! ${formatCurrency(winAmount)} (${multiplier}x) 🎊`;
                        resultMessage.innerHTML = `<span class="gold-gradient-text">${message}</span>`;
                        logType = 'bonus';
                        log(`🎰 BONUS: Ganador ${multiplier}x. Ganas ${formatCurrency(winAmount)}. Saldo: ${formatCurrency(balance)}`, logType);
                        
                        updateUI();
                        
                        setTimeout(() => {
                            resultMessage.innerHTML = `<span class="gold-text">¡Bonus completado! Vuelves a la ruleta normal</span>`;
                            setTimeout(() => {
                                deactivateBonusRound();
                                statusMessage.textContent = "¡Listo para otra ronda!";
                                statusMessage.className = "text-center font-semibold text-lg mb-4 h-6";
                                resultMessage.textContent = "";
                            }, 2000);
                        }, 2000);
                    } else {
                        message = `¡FELICIDADES! Ganas ${formatCurrency(winAmount)} (Multiplicador ${multiplier}x).`;
                        resultMessage.innerHTML = `<span class="text-green-400">${message}</span>`;
                        logType = 'win';
                        log(`Apuesta ${formatCurrency(currentBet)}. Ganador ${multiplier}x. Saldo: ${formatCurrency(balance)}`, logType);
                        statusMessage.textContent = "¡Listo para otra ronda!";
                        statusMessage.className = "text-center font-semibold text-lg mb-4 h-6";
                    }
                } else {
                    // Multiplicador 0x - Pierdes la apuesta
                    message = `¡MALA SUERTE! Has caído en 0x. Pierdes ${formatCurrency(currentBet)}.`;
                    resultMessage.innerHTML = `<span class="text-red-400">${message}</span>`;
                    logType = 'loss';
                    log(`Apuesta ${formatCurrency(currentBet)}. Resultado 0x. Saldo: ${formatCurrency(balance)}`, logType);
                    statusMessage.textContent = "¡Listo para otra ronda!";
                    statusMessage.className = "text-center font-semibold text-lg mb-4 h-6";
                }
                
                updateUI();

            }, duration + 50);
        }

        // Inicialización
        function init() {
            wheelSVGNormal.style.transform = 'rotate(0deg)';
            wheelSVGBonus.style.transform = 'rotate(0deg)';
            balanceDisplay.textContent = formatCurrency(balance);
            
            btnSpin.addEventListener('click', spinWheel);
            betInput.addEventListener('input', updateUI);
            
            updateUI();
            log("Juego de la Ruleta iniciado. ¡Suerte!", "info");
        }

        window.onload = init;
    </script>
