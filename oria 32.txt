        function formatCurrency(amount) {

        // LOGIC FOR SVG DRAWING (Dinámica)

        function polarToCartesian(radius, angleDeg) {
            const angleRad = (angleDeg - 90) * Math.PI / 180.0;
            return {
                x: radius * Math.cos(angleRad),
                y: radius * Math.sin(angleRad)
            };
        }

        function describeArc(R, startAngle, endAngle) {
            const start = polarToCartesian(R, endAngle);
            const end = polarToCartesian(R, startAngle);
            const largeArcFlag = endAngle - startAngle <= 180 ? "0" : "1";
            return [
                "M", 0, 0, 
                "L", start.x, start.y, 
                "A", R, R, 0, largeArcFlag, 0, end.x, end.y, 
                "Z" 
            ].join(" ");
        }

        /**
         * Genera los 32 sectores con los valores y textos correctos para la ruleta actual.
         * @param {object} wheelConfig - Configuración de la ruleta actual.
         */
        function generateWheelSectors(wheelConfig) {
            const fullSectors = [];
            const sectorDefs = BASE_SECTORS_DEFINITION;

            for (let i = 0; i < NUM_SECTORS; i++) {
                const baseSector = sectorDefs[i % sectorDefs.length];

                let finalMultiplier;
                let textContent;
                let isBonusSector = false;

                if (baseSector.id === 'bonus') {
                    isBonusSector = true;
                    textContent = wheelConfig.starChar;
                    finalMultiplier = wheelConfig.bonusMultiplier;
                    baseSector.color = '#eab308'; 
                } else {
                    finalMultiplier = baseSector.baseValue * wheelConfig.multiplierFactor;
                    textContent = finalMultiplier === 0 ? '0x' : `${finalMultiplier.toFixed(0)}x`;
                }

                const stopAngle = (i * SECTOR_ANGLE) + (SECTOR_ANGLE / 2);
                fullSectors.push({
                    ...baseSector, 
                    multiplier: finalMultiplier,
                    textContent: textContent,
                    stopAngle: stopAngle,
                    originalIndex: i,
                    isBonusSector: isBonusSector 
                });
            }
            return fullSectors;
        }

        /**
         * Draws the 32 sectors of the wheel and their text content.
         */
        function drawWheel() {
            sectorsContainer.innerHTML = '';
            SECTORS.forEach((sector, index) => { 
                const startAngle = index * SECTOR_ANGLE;
                const endAngle = (index + 1) * SECTOR_ANGLE;
                const centerAngle = startAngle + (SECTOR_ANGLE / 2);

                // 1. Dibuja el sector (Path)
                const path = document.createElementNS("http://www.w3.org/2000/svg", "path");
                path.setAttribute('d', describeArc(R, startAngle, endAngle));
                path.setAttribute('fill', sector.color);
                sectorsContainer.appendChild(path);

                // 2. Dibuja el texto del multiplicador (Text)
                const { x, y } = polarToCartesian(TEXT_RADIUS, centerAngle);
                const textElement = document.createElementNS("http://www.w3.org/2000/svg", "text");
                textElement.textContent = sector.textContent;

                textElement.setAttribute('x', x);
                textElement.setAttribute('y', y);
                textElement.setAttribute('dominant-baseline', 'middle'); 
                textElement.setAttribute('text-anchor', 'middle');

                let textRotation = centerAngle + 90;
                if (centerAngle > 90 && centerAngle < 270) {
                    textRotation += 180;
                }
                
                textElement.setAttribute('transform', `rotate(${textRotation}, ${x}, ${y})`);
                if (sector.isBonusSector) {
                    textElement.setAttribute('class', `sector-text !text-lg`); 
                } else {
                    textElement.setAttribute('class', `sector-text`);
                }

                sectorsContainer.appendChild(textElement);
            });
            
            // Actualizar el texto del centro para mostrar el nombre de la ruleta
            const currentConfig = WHEEL_CONFIGS[currentWheelIndex];
            document.querySelector('.wheel-center').textContent = currentConfig.name;
        }

        /**
         * Fills the rules table with sector information (Dinámica).
         */
        function fillRulesTable() {
            rulesBody.innerHTML = '';
            const currentConfig = WHEEL_CONFIGS[currentWheelIndex];
            
            const uniqueRules = {};
            SECTORS.forEach(sector => {
                let displayMultiplier = sector.multiplier;
                if (sector.isBonusSector) {
                    // Si es una estrella que AVANZA (Rueda 1 y 2), su valor mostrado es 0x
                    if (currentConfig.bonusMultiplier < 10000) {
                        displayMultiplier = 0; 
                    }
                }
                
                const key = sector.textContent;
                if (!uniqueRules[key]) {
                    uniqueRules[key] = {
                        textContent: sector.textContent,
                        color: sector.color,
                        totalProb: sector.totalProb,
                        probability: sector.probability,
                        multiplier: displayMultiplier
                    };
                }
            });
            
            const sortedRules = Object.values(uniqueRules).sort((a, b) => b.multiplier - a.multiplier);

            sortedRules.forEach(sector => {
                const row = rulesBody.insertRow();

                // Multiplier/Text
                let cell = row.insertCell();
                cell.textContent = sector.textContent;
                cell.classList.add('font-bold');

                // Color
                cell = row.insertCell();
                cell.innerHTML = `<span class="color-dot" style="background-color: ${sector.color};"></span>`;

                // Total Probability
                cell = row.insertCell();
                cell.textContent = (sector.totalProb * 100).toFixed(1) + '%';

                // Probability per Segment
                cell = row.insertCell();
                cell.textContent = (sector.probability * 100).toFixed(2) + '%';
            });
            
            // Actualizar el título de la tabla
            document.querySelector('.rules-table-title').textContent = 
                `Valores de los Sectores: ${currentConfig.name}`;
        }


        // ----------------------------------------------------
        // LOGIC FOR SELECTION AND STOP ANGLE CALCULATION
        // ----------------------------------------------------

        function selectWinningSector() {
            let totalWeight = SECTORS.reduce((sum, sector) => sum + sector.probability, 0);
            let randomNum = Math.random() * totalWeight;
            
            let cumulativeWeight = 0;
            for (const sector of SECTORS) {
                cumulativeWeight += sector.probability;
                if (randomNum < cumulativeWeight) {
                    return sector;
                }
            }
            return SECTORS[SECTORS.length - 1];
        }

        function calculateStopRotation(winningSector) {
            const sectorAngle = winningSector.stopAngle;
            let angleToAlign = 360 - sectorAngle; 

            const minFullSpins = 5;
            const fullRotation = minFullSpins * 360; 

            const startRotation = lastStopRotation % 360;
            let rotationDelta = angleToAlign - startRotation;
            if (rotationDelta < 0) {
                rotationDelta += 360;
            }
            
            const jitter = (Math.random() - 0.5) * (SECTOR_ANGLE * 0.4);
            
            const totalRotation = fullRotation + rotationDelta + jitter;
            return totalRotation;
        }

        /**
         * Obtiene la rotación actual del SVG para un giro suave.
         */
        function getCurrentRotation() {
            const style = window.getComputedStyle(wheelSVG);
            const transform = style.getPropertyValue('transform');

            if (transform === 'none') return 0;
            const matrix = transform.match(/matrix\(([^,]+),([^,]+),([^,]+),([^,]+),([^,]+),([^,]+)\)/);
            if (matrix) {
                const a = parseFloat(matrix[1]);
                const b = parseFloat(matrix[2]);
                const angleRad = Math.atan2(b, a);
                let angleDeg = angleRad * (180 / Math.PI);
                
                if (angleDeg < 0) {
                    angleDeg += 360;
                }
                return angleDeg;
            }
            return 0;
        }

        /**
         * Restaura el estado visual de la ruleta después de un giro.
         */
        function resetWheelAndCheckBalance() {
            updateBalanceDisplay();
            
            // 1. Limpiar la transición CSS para evitar saltos al reanudar la animación de espera
            wheelSVG.style.transition = `transform 0s`;
            // 2. Colocar la ruleta en el ángulo de parada (sólo la parte 0-360) para la animación de espera
            wheelSVG.style.transform = `rotate(${lastStopRotation % 360}deg)`; 
            
            // 3. Volvemos a activar la animación de espera CSS
            wheelSVG.style.animation = 'subtle-spin 60s linear infinite';
            
            isSpinning = false;
            
            if (balance <= 0) {
                 btnSpin.disabled = true;
                 statusMessage.textContent = '¡Juego terminado! Te has quedado sin saldo.';
                 log('El juego ha terminado por falta de saldo.', 'loss');
            } else {
                checkInput(); // Re-chequear para habilitar el botón si el saldo es suficiente
                statusMessage.textContent = 'Giro completado. Introduce la siguiente apuesta.';
            }
        }


        function updateBalanceDisplay() {
            balanceDisplay.textContent = formatCurrency(balance);
            if (balance < 0) {
                balanceDisplay.classList.remove('text-green-500');
                balanceDisplay.classList.add('text-red-500');
            } else {
                balanceDisplay.classList.remove('text-red-500');
                balanceDisplay.classList.add('text-green-500');
            }
        }

        function checkInput() {
            const betInputEl = document.getElementById('bet-input');
            const betValue = parseInt(betInputEl.value, 10);
            const currentWheelName = WHEEL_CONFIGS[currentWheelIndex].name;
            
            // Si no estamos en la ruleta base, el botón debe decir "GIRAR GRATIS" y la apuesta no importa
            if (currentWheelIndex > 0) {
                betInputEl.disabled = true;
                btnSpin.textContent = 'GIRAR GRATIS';
                // Si ya está girando, deshabilitar, si no, habilitar
                btnSpin.disabled = isSpinning; 
                statusMessage.textContent = `¡BONO ACTIVO! Estás en la ${currentWheelName}. Gira para continuar.`;
                statusMessage.classList.remove('text-red-400', 'text-gray-400');
                statusMessage.classList.add('text-yellow-400');
                return;
            }
            
            // Lógica normal para la Ruleta Base
            betInputEl.disabled = false;
            btnSpin.textContent = 'GIRAR';
            
            const isValid = betValue > 0 && betValue <= balance && !isSpinning;
            btnSpin.disabled = !isValid;

            if (betValue <= 0) {
                statusMessage.textContent = 'La apuesta debe ser superior a 0.';
                statusMessage.classList.add('text-red-400');
            } else if (betValue > balance) {
                statusMessage.textContent = 'Apuesta demasiado alta. Saldo insuficiente.';
                statusMessage.classList.add('text-red-400');
            } else {
                statusMessage.textContent = 'Apuesta lista. Presiona "GIRAR".';
                statusMessage.classList.remove('text-red-400', 'text-yellow-400');
                statusMessage.classList.add('text-gray-400');
            }
        }
        
        /**
         * Realiza la transición a la siguiente ruleta de bonificación.
         */
        function startBonusSpin() {
            const nextIndex = WHEEL_CONFIGS[currentWheelIndex].nextWheelIndex;
            currentWheelIndex = nextIndex;
            
            // 1. Redibujar la ruleta con la nueva configuración
            SECTORS = generateWheelSectors(WHEEL_CONFIGS[currentWheelIndex]);
            drawWheel(); 
            fillRulesTable();

            // 2. Reiniciar estado para el giro de bonificación
            isSpinning = false;
            checkInput(); // Esto habilitará el botón con el texto 'GIRAR GRATIS'
            
            // Limpiar resultados anteriores
            multiplierDisplay.textContent = '';
            prizeDisplay.textContent = '';
        }

        async function spin() {
            const bet = parseInt(betInput.value, 10);
            const isBonusSpin = currentWheelIndex > 0;
            
            // Validaciones
            if (!isBonusSpin && (bet <= 0 || bet > balance || isSpinning)) return;
            if (isBonusSpin && isSpinning) return;

            // 1. Preparar el estado y la interfaz
            isSpinning = true;
            btnSpin.disabled = true;
            statusMessage.textContent = `¡La ${WHEEL_CONFIGS[currentWheelIndex].name} está girando...!`;
            multiplierDisplay.textContent = '';
            prizeDisplay.textContent = '...';
            prizeDisplay.className = 'text-gray-200 text-4xl';

            // 1.1 Detener animación de espera y obtener el ángulo de inicio
            wheelSVG.style.animation = 'none'; 
            const currentAngle = getCurrentRotation(); 
            wheelSVG.style.transform = `rotate(${currentAngle}deg)`;

            // 2. Restar apuesta del saldo (Solo si no es giro de bono)
            if (!isBonusSpin) {
                balance -= bet;
                updateBalanceDisplay();
                log(`Apuesta de ${formatCurrency(bet)} deducida.`, 'info');
            } else {
                log(`Giro GRATIS en ${WHEEL_CONFIGS[currentWheelIndex].name}. Apuesta base: ${formatCurrency(bet)}`, 'info');
            }

            // 3. Seleccionar el ganador y calcular la rotación
            const winningSector = selectWinningSector();
            const totalRotationOffset = calculateStopRotation(winningSector);
            
            // 4. Aplicar el giro con CSS Transition (rápido)
            const spinDuration = 6000;
            wheelSVG.style.transition = `transform ${spinDuration / 1000}s cubic-bezier(0.1, 0.5, 0.2, 1)`;
            wheelSVG.style.transform = `rotate(${totalRotationOffset}deg)`;
            lastStopRotation = totalRotationOffset;

            // 5. Esperar a que termine la animación
            await new Promise(resolve => setTimeout(resolve, spinDuration + 100));

            // 6. Determinar el resultado y actualizar el saldo
            const multiplier = winningSector.multiplier;
            const isAdvanceBonus = winningSector.isBonusSector && multiplier < 10000; // Es estrella PERO no es el premio máximo

            // --- Lógica de la Cascada ---
            if (isAdvanceBonus) {
                // Caída en el sector Estrella de BONIFICACIÓN (Ruletas 1 o 2)
                
                multiplierDisplay.textContent = `¡BONUS ${winningSector.textContent}!`;
                prizeDisplay.textContent = `¡Próxima Ruleta ${WHEEL_CONFIGS[currentWheelIndex].nextWheelIndex + 1}!`;
                prizeDisplay.classList.remove('loss-text', 'text-gray-200');
                prizeDisplay.classList.add('text-yellow-400');
                
                log(`¡Ganas un bono! Avanzas a Ruleta ${WHEEL_CONFIGS[currentWheelIndex].nextWheelIndex + 1} (Giro Gratis).`, 'win');

                // 7. Transición al siguiente nivel (Giro GRATIS)
                startBonusSpin();
                
                // ** LÓGICA DE AUTO-CASCADA: Resetea el estado visual e inicia el siguiente giro
                resetWheelAndCheckBalance(); 

                // 8. Ejecuta el siguiente giro automáticamente después de un retraso
                setTimeout(() => {
                    // Llama a spin() para el giro gratis en la nueva ruleta
                    spin(); 
                }, 2500); // 2.5 segundos para mostrar el cambio de ruleta.
                
                return; // Termina la función spin aquí para el giro de cascada.
                
            } else {
                // Multiplicador de premio normal (incluida la estrella de la Rueda 3, que es 10000x)
                
                const winnings = bet * multiplier;
                balance += winnings;

                multiplierDisplay.textContent = `Multiplicador Obtenido: ${winningSector.textContent}`;
                
                if (multiplier > 1) {
                    const profit = winnings - (isBonusSpin ? 0 : bet);
                    
                    prizeDisplay.textContent = `¡GANAS ${formatCurrency(profit)}!`;
                    prizeDisplay.classList.remove('loss-text', 'text-gray-200');
                    prizeDisplay.classList.add('win-text');
                    
                    log(`Apuesta de ${formatCurrency(bet)}, ¡Premio de ${formatCurrency(winnings)}! (Ganancia Neta: ${formatCurrency(profit)})`, 'win');
                } else if (multiplier === 1.0) {
                    prizeDisplay.textContent = `Recuperas tu apuesta.`;
                    prizeDisplay.classList.remove('win-text', 'loss-text');
                    prizeDisplay.classList.add('text-gray-200');
                    log(`Apuesta de ${formatCurrency(bet)}, recuperas la apuesta.`, 'info');
                } else {
                    const netLoss = isBonusSpin ? 0 : bet;
                    prizeDisplay.textContent = `PIERDES ${isBonusSpin ? 'el giro de bono.' : formatCurrency(netLoss)}`;
                    prizeDisplay.classList.remove('win-text', 'text-gray-200');
                    prizeDisplay.classList.add('loss-text');
                    log(`Apuesta de ${formatCurrency(bet)}, Pierdes -${formatCurrency(netLoss)}.`, 'loss');
                }

                // 9. Volver a la Ruleta Base si no es la Ruleta Base (Fin de cascada o giro normal)
                if (currentWheelIndex > 0) {
                    currentWheelIndex = 0; // Vuelve a la ruleta base
                    SECTORS = generateWheelSectors(WHEEL_CONFIGS[currentWheelIndex]);
                    drawWheel(); 
                    fillRulesTable();
                }

                // 10. Reiniciar estado (Cleanup final para giros normales)
                resetWheelAndCheckBalance();
            }
        }

        // ----------------------------------------------------
        // EVENT LISTENERS AND INIT
        // ----------------------------------------------------

        btnSpin.addEventListener('click', spin);
        betInput.addEventListener('input', checkInput);

        function init() {
            drawWheel();
            fillRulesTable();
            updateBalanceDisplay();
            checkInput();
            log('Ruleta de la Suerte lista. ¡Buena suerte!', 'info');
        }

        // Inicializar la aplicación
        window.onload = init;
    </script>
