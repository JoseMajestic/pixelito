  <script>
    // ---------------------------
    // MODELO DE CARTAS
    // ---------------------------
    const ranks = [
      { value: 2, label: "2" },
      { value: 3, label: "3" },
      { value: 4, label: "4" },
      { value: 5, label: "5" },
      { value: 6, label: "6" },
      { value: 7, label: "7" },
      { value: 8, label: "8" },
      { value: 9, label: "9" },
      { value: 10, label: "10" },
      { value: 11, label: "J" },
      { value: 12, label: "Q" },
      { value: 13, label: "K" },
      { value: 14, label: "A" }
    ];
    const suits = [
      { symbol: "♥", colorClass: "red" },
      { symbol: "♦", colorClass: "red" },
      { symbol: "♣", colorClass: "black" },
      { symbol: "♠", colorClass: "black" }
    ];
    function drawCard() {
      const r = ranks[Math.floor(Math.random() * ranks.length)];
      const s = suits[Math.floor(Math.random() * suits.length)];
      return {
        value: r.value,
        label: r.label,
        suit: s.symbol,
        colorClass: s.colorClass
      };
    }

    // ---------------------------
    // ESTADO
    // ---------------------------
    let balance = 1000;
    const START_BALANCE = 1000;
    let currentCard = null;
    let nextCard = null;
    let roundId = 1;
    let inDecision = false;

    // --- VARIABLES PARA RACHAS ---
    let currentStreakMultiplier = 1.0; // Multiplicador acumulado
    let currentBet = 0; // Apuesta inicial de la racha
    // -----------------------------

    const balanceDisplay = document.getElementById("balanceDisplay");
    const statusText = document.getElementById("statusText");
    const roundInfo = document.getElementById("roundInfo");
    const lastResultEl = document.getElementById("lastResult");
    const logElement = document.getElementById("log");
    const currentCardEl = document.getElementById("currentCard");
    const nextCardEl = document.getElementById("nextCard");

    const betInput = document.getElementById("betInput");
    const btnHigher = document.getElementById("btnHigher");
    const btnLower = document.getElementById("btnLower");
    const btnNewRound = document.getElementById("btnNewRound");
    const btnCashout = document.getElementById("btnCashout"); 

    const tagHigher = document.getElementById("tagHigher");
    const tagLower = document.getElementById("tagLower");
    const tagCashout = document.getElementById("tagCashout"); 
    const metricMult = document.getElementById("metricMult");

    // ---------------------------
    // UTILS
    // ---------------------------
    function formatCurrency(value) {
      return value.toLocaleString("es-ES", {
        style: "currency",
        currency: "EUR",
        maximumFractionDigits: 0
      });
    }

    function log(message, type = "system") {
      const entry = document.createElement("div");
      entry.classList.add("log-entry");
      if (type === "win") entry.classList.add("win");
      if (type === "loss") entry.classList.add("loss");
      const label = document.createElement("span");
      label.classList.add("label");
      label.textContent =
        type === "win" ? "WIN" : type === "loss" ?
        "LOSS" : "SYS";
      const text = document.createElement("span");
      text.textContent = " " + message;
      entry.appendChild(label);
      entry.appendChild(text);
      logElement.prepend(entry);
    }

    function updateBalance() {
      balanceDisplay.textContent = formatCurrency(balance);
    }

    function setStatus(text) {
      statusText.textContent = text;
    }

    function setButtonsEnabled(enabled) {
      const buttons = [btnHigher, btnLower];
      buttons.forEach((btn) => {
        if (enabled) {
          btn.classList.remove("disabled");
          btn.disabled = false;
        } else {
          btn.classList.add("disabled");
          btn.disabled = true;
        }
      });
    }

    function renderCard(card, element, hidden = false) {
      element.classList.remove("hidden");
      const rankCorners = element.querySelectorAll(".rank");
      const center = element.querySelector(".center");
      const suitsInside = element.querySelectorAll(".suit");
      element.classList.remove("red", "black");
      
      if (hidden) {
        element.classList.add("hidden");
        return;
      }

      element.classList.add(card.colorClass);
      rankCorners.forEach((node) => {
        node.textContent = card.label;
      });
      suitsInside.forEach((node) => {
        node.textContent = card.suit;
      });
      if (center) {
        center.textContent = card.suit;
      }
    }

    // ---------------------------
    // PROBABILIDADES Y MULTIS
    // ---------------------------
    function calculateProbabilities() {
      if (!currentCard) return { higherProb: 0.5, lowerProb: 0.5 };
      const v = currentCard.value;
      const total = ranks.length; // 13 cartas (del 2 al A/14)

      // Probabilidad de Mayor o Igual: (14 - Valor Actual + 1) / 13
      const higherOrEqual = (ranks[ranks.length - 1].value - v + 1) / total;
      // Probabilidad de Menor o Igual: (Valor Actual - 2 + 1) / 13
      const lowerOrEqual = (v - ranks[0].value + 1) / total;
      
      return {
        higherProb: higherOrEqual,
        lowerProb: lowerOrEqual
      };
    }

    function calculateMultipliers() {
      const { higherProb, lowerProb } = calculateProbabilities();
      const houseEdge = 0.96; // Margen de la casa teórico (4% para la demo)

      const higherMulti = houseEdge / higherProb;
      const lowerMulti = houseEdge / lowerProb;
      
      // Asegurarse de que el multiplicador mínimo sea 1.01x
      return {
        higher: Math.max(1.01, parseFloat(higherMulti.toFixed(2))),
        lower: Math.max(1.01, parseFloat(lowerMulti.toFixed(2)))
      };
    }

    function updateMultipliersUI() {
      const { higher, lower } = calculateMultipliers();
      tagHigher.textContent = higher.toFixed(2) + "x";
      tagLower.textContent = lower.toFixed(2) + "x";
      metricMult.textContent =
        "H " + higher.toFixed(2) + "x · L " + lower.toFixed(2) + "x";
    }
    
    // Función para actualizar el botón de Cashout (NUEVA)
    function updateCashoutUI() {
        const cashoutMult = currentStreakMultiplier;
        // Calcular la ganancia total (apuesta inicial * multiplicador acumulado)
        const potentialWin = Math.round(currentBet * cashoutMult);
        
        // El texto del botón ahora muestra la ganancia potencial total
        btnCashout.innerHTML = `Retirarse &mdash; <span id="tagCashout">${formatCurrency(potentialWin)}</span>`;
        
        // Si el multiplicador es 1.0 (o menos por error), se inhabilita porque no hay ganancias acumuladas
        if (cashoutMult <= 1.0) {
            btnCashout.classList.add("disabled");
            btnCashout.disabled = true;
        } else {
            btnCashout.classList.remove("disabled");
            btnCashout.disabled = false;
        }
    }
    
    // Función para resetear el estado de la ronda (NUEVA)
    function resetRoundState() {
        currentStreakMultiplier = 1.0;
        currentBet = 0;
        inDecision = false;
        
        setButtonsEnabled(false);
        btnCashout.classList.add("disabled");
        btnCashout.disabled = true;
        btnCashout.innerHTML = `Retirarse &mdash; <span id="tagCashout">0 €</span>`;
        
        btnNewRound.disabled = false; // Habilitar botón para empezar nueva racha
        btnNewRound.classList.remove("disabled");
        betInput.disabled = false;
        
        updateMultipliersUI();
        updateBalance();
        
        // Dibujar una nueva carta neutral para empezar la ronda, y ocultar la siguiente
        currentCard = drawCard(); 
        renderCard(currentCard, currentCardEl, false);
        renderCard(currentCard, nextCardEl, true);
        
        roundInfo.textContent = "";
        lastResultEl.textContent = "Listo para una nueva racha.";
        setStatus("Nueva ronda lista. Configura una apuesta y pulsa 'Empezar racha'.");
    }

    // ---------------------------
    // FLOW PRINCIPAL
    // ---------------------------
    
    // La función newRound ahora INICIA LA RACHA y cobra la apuesta inicial
    function newRound() {
        if(inDecision) { 
            setStatus("Ya estás en una racha. Elige mayor/menor o retírate.");
            return;
        }

        // 1. Validar y tomar apuesta
        currentBet = Number(betInput.value) || 0;
        if (currentBet <= 0) {
            setStatus("Configura una apuesta válida para empezar.");
            log("Apuesta inválida.", "system");
            return;
        }
        if (currentBet > balance) {
            setStatus("Saldo insuficiente para la apuesta inicial.");
            log("Intento de apuesta sin saldo suficiente.", "system");
            return;
        }

        // 2. Establecer variables de racha y cobrar apuesta
        currentStreakMultiplier = 1.0;
        balance -= currentBet; // Se descuenta la apuesta inicial
        updateBalance();

        // 3. Flujo normal de inicio
        currentCard = drawCard();
        nextCard = null;
        inDecision = true;

        renderCard(currentCard, currentCardEl, false);
        renderCard(currentCard, nextCardEl, true); // oculta

        updateMultipliersUI();
        updateCashoutUI(); 

        setButtonsEnabled(true);
        btnNewRound.disabled = true; 
        btnNewRound.classList.add("disabled");
        betInput.disabled = true; 

        setStatus(`Ronda ${roundId}. Apuesta: ${formatCurrency(currentBet)}. ¡A jugar!`);
        roundInfo.textContent = `Racha #${roundId} | Apuesta: ${formatCurrency(currentBet)}`;
        lastResultEl.textContent = `Racha ${roundId} en progreso (x${currentStreakMultiplier.toFixed(2)}).`;
        roundId += 1;
    }

    // La función resolveBet ahora CONTINÚA LA RACHA o la termina
    function resolveBet(direction) {
      if (!inDecision) return;

      const bet = currentBet;
      if (bet <= 0) return; 

      inDecision = false;
      setButtonsEnabled(false);
      btnCashout.disabled = true; 
      btnCashout.classList.add("disabled");

      // Animación de "revelar"
      setStatus("Revelando carta...");
      setTimeout(() => {
        nextCard = drawCard();
        renderCard(nextCard, nextCardEl, false); // Revela la carta

        const { higher, lower } = calculateMultipliers();
        const chosenMult = direction === "higher" ? higher : lower;

        let win = false;
        if (direction === "higher") {
          win = nextCard.value >= currentCard.value;
        } else {
          win = nextCard.value <= currentCard.value;
        }

        if (win) {
          // --- GANA: CONTINÚA LA RACHA ---
          currentStreakMultiplier *= chosenMult; // Acumular multiplicador

          // La próxima carta se convierte en la actual para la siguiente decisión
          currentCard = nextCard; 

          // Preparar para la siguiente sub-ronda (pequeño delay visual)
          setTimeout(() => {
            renderCard(currentCard, currentCardEl, false);
            renderCard(currentCard, nextCardEl, true); // Oculta de nuevo
            updateMultipliersUI();
            updateCashoutUI();

            inDecision = true;
            setButtonsEnabled(true);
            btnCashout.disabled = false; 
            btnCashout.classList.remove("disabled");

            setStatus(
                `¡Acierto! Multiplicador total: x${currentStreakMultiplier.toFixed(2)}. Elige de nuevo o retírate.`
            );
            roundInfo.textContent = `Racha # ${roundId-1} | Multiplicador: x${currentStreakMultiplier.toFixed(2)}`;
            lastResultEl.textContent = `Carta: ${currentCard.label}${currentCard.suit}. Ganancia potencial: ${formatCurrency(currentBet * currentStreakMultiplier)}.`;
            log(`Acierto: ${direction === "higher" ? "mayor/igual" : "menor/igual"} (x${chosenMult.toFixed(2)}). Total x${currentStreakMultiplier.toFixed(2)}.`, "win");
          }, 500);
          
        } else {
          // --- PIERDE: FIN DE LA RACHA ---
          const finalLoss = currentBet;
          setStatus("¡Fallo! La racha ha terminado. Pierdes la apuesta inicial.");
          lastResultEl.textContent = `Fallo: -${formatCurrency(finalLoss)} de tu apuesta inicial. Carta: ${nextCard.label}${nextCard.suit}.`;
          log(
              `Fallo. Racha terminada con carta ${nextCard.label}${nextCard.suit}. Apuesta inicial perdida: ${formatCurrency(finalLoss)}.`,
              "loss"
          );
          
          resetRoundState(); // Reiniciar estado
        }
      }, 260);
    }
    
    // Función para manejar el Cashout (NUEVA)
    function doCashout() {
        if (!inDecision || currentStreakMultiplier <= 1.0) return;

        const totalWin = Math.round(currentBet * currentStreakMultiplier);
        balance += totalWin; // Sumar apuesta inicial + todas las ganancias
        updateBalance();

        setStatus(`¡Cashout exitoso! Ganancia total de ${formatCurrency(totalWin)} (x${currentStreakMultiplier.toFixed(2)}).`);
        lastResultEl.textContent = `Cashout: Ganancia total de ${formatCurrency(totalWin)}.`;
        log(
            `Cashout de apuesta ${formatCurrency(currentBet)}. Ganancia final ${formatCurrency(
                totalWin
            )} (x${currentStreakMultiplier.toFixed(2)}).`,
            "win"
        );

        resetRoundState(); // Reiniciar estado
    }


    // ---------------------------
    // EVENTOS
    // ---------------------------
    btnNewRound.addEventListener("click", () => {
      newRound();
    });
    
    btnCashout.addEventListener("click", () => {
        if (btnCashout.disabled) return;
        doCashout();
    }); 
    
    // Eventos para elegir Mayor/Menor (inicia la ronda si no ha comenzado)
    btnHigher.addEventListener("click", () => {
        if (!inDecision) newRound();
        if (btnHigher.disabled) return;
        resolveBet("higher");
    });
    
    btnLower.addEventListener("click", () => {
        if (!inDecision) newRound();
        if (btnLower.disabled) return;
        resolveBet("lower");
    });


    // ---------------------------
    // INIT
    // ---------------------------
    function init() {
      // Inicializar balance en pantalla
      document.getElementById("metricStart").textContent = formatCurrency(START_BALANCE);
      balance = START_BALANCE;
      
      // Empezar en estado de espera para nueva apuesta
      resetRoundState(); 
    }

    init();
  </script>
