          const x = (col * cellWidth) + (cellWidth / 2);
          const y = (row * cellHeight) + (cellHeight / 2);

          } else {
            d += ` L ${x} ${y}`;
          }

        path.setAttribute("stroke", getLineColor(lineIndex));
        linesOverlay.appendChild(path);
      });

    function fillGrid() {
      for (let r = 0; r < CONFIG.ROWS; r++) {
        for (let c = 0; c < CONFIG.COLS; c++) {
          // Probabilidad reducida para símbolo Wheel (5%)
          let symbolIndex;
          if (Math.random() < 0.05) {
            // 5% de probabilidad para símbolo Wheel
            symbolIndex = SYMBOLS.findIndex(s => s.type === "wheel");
          } else {
            symbolIndex = Math.floor(Math.random() * SYMBOLS.length);
          }
          
          grid[r][c].symbolIndex = symbolIndex;
          grid[r][c].element.classList.remove("win", "wild", "scatter", "wheel", "spinning");
          
          const symbol = SYMBOLS[symbolIndex];
          grid[r][c].span.textContent = symbol.icon;
          
          if (symbol.type === "wild") {
            grid[r][c].element.classList.add("wild");
          } else if (symbol.type === "scatter") {
            grid[r][c].element.classList.add("scatter");
          } else if (symbol.type === "wheel") {
            grid[r][c].element.classList.add("wheel");
          }
        }
      }
    }

    function animateSpin() {
      return new Promise(resolve => {
        const reelElements = document.querySelectorAll('.cell');
        reelElements.forEach(cell => {
          cell.classList.add('spinning');
        });
        
        let spins = 0;
        const spinInterval = setInterval(() => {
          fillGrid();
          spins++;
          
          if (spins > 15) {
            clearInterval(spinInterval);
            reelElements.forEach(cell => {
              cell.classList.remove('spinning');
            });
            resolve();
          }
        }, 60);
      });
    }

    // ====================
    // LÓGICA DE PAGO
    // ====================
    function evaluatePaylines() {
      const winningLines = [];
      let totalWin = 0;
      let scatterCount = 0;
      let hasWheelSymbol = false;
      
      // Contar scatters y símbolos wheel
      for (let r = 0; r < CONFIG.ROWS; r++) {
        for (let c = 0; c < CONFIG.COLS; c++) {
          const symbol = SYMBOLS[grid[r][c].symbolIndex];
          if (symbol.type === "scatter") {
            scatterCount++;
          }
          if (symbol.type === "wheel") {
            hasWheelSymbol = true;
          }
        }
      }
      
      // Evaluar líneas de pago
      for (let lineIndex = 0; lineIndex < activeLines; lineIndex++) {
        const line = PAYLINES[lineIndex];
        let firstSymbol = null;
        let consecutive = 0;
        let hasWild = false;
        
        for (let pos = 0; pos < line.length; pos++) {
          const [col, row] = line[pos];
          const symbol = SYMBOLS[grid[row][col].symbolIndex];
          
          if (symbol.type === "wild") {
            hasWild = true;
            consecutive++;
            if (firstSymbol === null && pos > 0) {
              continue;
            }
          } else if (firstSymbol === null || 
                    (symbol.type === "regular" && 
                     (firstSymbol.name === symbol.name || hasWild))) {
            if (firstSymbol === null) {
              firstSymbol = symbol;
            }
            consecutive++;
            hasWild = false;
          } else {
            break;
          }
        }
        
        // Verificar combinación ganadora
        if (consecutive >= 3 && firstSymbol && firstSymbol.type === "regular") {
          const multiplier = firstSymbol.multiplier[consecutive - 1] || 0;
          if (multiplier > 0) {
            const win = currentBet * multiplier;
            totalWin += win;
            winningLines.push({
              lineIndex,
              symbol: firstSymbol.name,
              count: consecutive,
              multiplier,
              win
            });
          }
        }
      }
      
      // Pago de scatters
      if (scatterCount >= 3) {
        const scatterSymbol = SYMBOLS.find(s => s.type === "scatter");
        const multiplier = scatterSymbol.multiplier[scatterCount - 1] || 0;
        if (multiplier > 0) {
          const scatterWin = (currentBet * activeLines) * multiplier;
          totalWin += scatterWin;
          
          winningLines.push({
            lineIndex: -1,
            symbol: "Scatter",
            count: scatterCount,
            multiplier,
            win: scatterWin
          });
        }
        
        // Giros gratis
        if (freeSpinsRemaining === 0) {
          freeSpinsRemaining = CONFIG.FREE_SPINS_ON_SCATTER;
          log(`¡${scatterCount} Scatters! ${freeSpinsRemaining} giros gratis activados.`, "bonus");
        }
      }
      
      return { winningLines, totalWin, scatterCount, hasWheelSymbol };
    }

    function highlightWinningSymbols(winningLines) {
      document.querySelectorAll('.cell').forEach(cell => {
        cell.classList.remove('win');
      });
      
      winningLines.forEach(line => {
        if (line.lineIndex >= 0) {
          const linePath = PAYLINES[line.lineIndex];
          linePath.forEach(([col, row]) => {
            grid[row][col].element.classList.add('win');
          });
        } else {
          // Scatters
          for (let r = 0; r < CONFIG.ROWS; r++) {
            for (let c = 0; c < CONFIG.COLS; c++) {
              if (SYMBOLS[grid[r][c].symbolIndex].type === "scatter") {
                grid[r][c].element.classList.add('win');
              }
            }
          }
        }
      });
    }

    // ====================
    // SISTEMA DE LA NORIA
    // ====================
    function createWheelSectors() {
      wheelElement.innerHTML = '';
      wheelSectors = [];
      
      const sectorAngle = 360 / WHEEL_CONFIG.SECTORS;
      const multipliers = [...WHEEL_CONFIG.MULTIPLIERS]
        .sort(() => Math.random() - 0.5)
        .slice(0, WHEEL_CONFIG.SECTORS);
      
      for (let i = 0; i < WHEEL_CONFIG.SECTORS; i++) {
        const sector = document.createElement('div');
        sector.classList.add('wheel-sector');
        sector.style.transform = `rotate(${i * sectorAngle}deg)`;
        
        const multiplier = multipliers[i];
        sector.innerHTML = `<span>x${multiplier}</span>`;
        
        wheelElement.appendChild(sector);
        wheelSectors.push({
          element: sector,
          multiplier: multiplier,
          angle: i * sectorAngle
        });
      }
    }

    function checkWheelActivation(winningLines, totalWin) {
      if (wheelActive) return false;
      
      // Condiciones para activar la noria:
      // 1. Al menos una línea ganadora
      // 2. Ganancia mayor al mínimo
      // 3. Probabilidad del 15%
      if (winningLines.length > 0 && 
          totalWin >= WHEEL_CONFIG.MIN_WIN_FOR_WHEEL &&
          Math.random() < WHEEL_CONFIG.APPEAR_CHANCE) {
        
        // Encontrar la línea con mayor ganancia para multiplicar
        const biggestWinLine = winningLines.reduce((max, line) => 
          line.win > max.win ? line : max, winningLines[0]);
        
        return {
          activate: true,
          baseWin: biggestWinLine.win,
          lineIndex: biggestWinLine.lineIndex
        };
      }
      
      return { activate: false };
    }

    function activateWheel(baseWinAmount, lineIndex) {
      wheelActive = true;
      wheelBaseWin = baseWinAmount;
      wheelMultiplier = 1;
      wheelCountdown = WHEEL_CONFIG.COUNTDOWN_SECONDS;
      
      // Actualizar UI de la noria
      baseWinAmountElement.textContent = formatCurrency(baseWinAmount);
      wheelMultiplierElement.textContent = 'x1';
      totalWinAmountElement.textContent = formatCurrency(baseWinAmount);
      
      // Crear sectores de la noria
      createWheelSectors();
      
      // Mostrar modal
      wheelModal.classList.add('active');
      
      // Iniciar countdown
      spinWheelButton.disabled = true;
      spinWheelButton.innerHTML = '<span class="icon">🎡</span><span>Girando en 3...</span>';
      
      const countdownInterval = setInterval(() => {
        wheelCountdown--;
        wheelCountdownElement.textContent = wheelCountdown;
        
        if (wheelCountdown <= 0) {
          clearInterval(countdownInterval);
          wheelCountdownElement.style.display = 'none';
          
          // Habilitar botón de girar (pero girará automáticamente)
          spinWheelButton.disabled = false;
          spinWheelButton.textContent = '¡Girando!';
          
          // Girar automáticamente después de 1 segundo
          setTimeout(spinWheel, 1000);
        } else {
          spinWheelButton.innerHTML = `<span class="icon">🎡</span><span>Girando en ${wheelCountdown}...</span>`;
        }
      }, 1000);
      
      log(`🎡 ¡Noria activada! Ganancia base: ${formatCurrency(baseWinAmount)}`, 'wheel');
    }

    function spinWheel() {
      const spinButton = document.getElementById('spinWheelButton');
      spinButton.disabled = true;
      spinButton.textContent = '¡Girando!';
      
      // Número aleatorio de giros completos
      const fullRotations = 4 + Math.floor(Math.random() * 5);
      
      // Sector aleatorio donde parar
      const stopSector = Math.floor(Math.random() * WHEEL_CONFIG.SECTORS);
      const sectorAngle = 360 / WHEEL_CONFIG.SECTORS;
      
      // Ángulo final
      const finalAngle = (fullRotations * 360) + (stopSector * sectorAngle);
      
      // Multiplicador del sector seleccionado
      wheelMultiplier = wheelSectors[stopSector].multiplier;
      
      // Animación de giro
      wheelElement.style.transition = `transform ${WHEEL_CONFIG.ANIMATION_DURATION}ms cubic-bezier(0.2, 0.8, 0.3, 1)`;
      wheelElement.style.transform = `rotate(${finalAngle}deg)`;
      
      // Efecto de desaceleración
      setTimeout(() => {
        wheelElement.style.transition = 'transform 0.8s ease-out';
      }, WHEEL_CONFIG.ANIMATION_DURATION * 0.7);
      
      // Cuando termine la animación
      setTimeout(() => {
        // Calcular ganancia total
        const totalWin = wheelBaseWin * wheelMultiplier;
        
        // Actualizar UI
        wheelMultiplierElement.textContent = `x${wheelMultiplier}`;
        totalWinAmountElement.textContent = formatCurrency(totalWin);
        
        // Efectos visuales
        wheelMultiplierElement.classList.add('flash');
        wheelElement.classList.add('flash');
        
        // Añadir ganancia al balance
        balance += totalWin;
        lastWin = totalWin;
        updateUI();
        
        // Crear confeti
        createConfetti();
        
        // Log
        log(`🎡 Multiplicador: x${wheelMultiplier}! Ganancia: ${formatCurrency(totalWin)}`, 'wheel');
        
        // Habilitar cierre después de 3 segundos
        setTimeout(() => {
          spinButton.textContent = '¡Recolectar!';
          spinButton.onclick = collectWheelWin;
          spinButton.disabled = false;
        }, 3000);
        
      }, WHEEL_CONFIG.ANIMATION_DURATION + 800);
    }

    function collectWheelWin() {
      wheelModal.classList.remove('active');
      
      // Resetear
      setTimeout(() => {
        wheelActive = false;
        const spinButton = document.getElementById('spinWheelButton');
        spinButton.disabled = false;
        spinButton.textContent = 'Girar Rueda';
        spinButton.onclick = null;
        spinButton.innerHTML = '<span class="icon">🎡</span><span>Girar Rueda</span>';
        
        wheelElement.style.transition = 'none';
        wheelElement.style.transform = 'rotate(0deg)';
        wheelCountdownElement.style.display = 'block';
        wheelCountdownElement.textContent = '3';
        
        // Continuar con el juego normal
        setSpinEnabled(true);
      }, 300);
    }

    function createConfetti() {
      const confettiCount = 150;
      const colors = ['#ef4444', '#f59e0b', '#84cc16', '#10b981', '#06b6d4', '#3b82f6', '#8b5cf6', '#d946ef', '#ec4899'];
      
      for (let i = 0; i < confettiCount; i++) {
        const confetti = document.createElement('div');
        confetti.classList.add('confetti');
        
        // Posición y estilo aleatorio
        confetti.style.left = Math.random() * 100 + 'vw';
        confetti.style.width = Math.random() * 20 + 5 + 'px';
        confetti.style.height = Math.random() * 20 + 5 + 'px';
        confetti.style.backgroundColor = colors[Math.floor(Math.random() * colors.length)];
        confetti.style.borderRadius = Math.random() > 0.5 ? '50%' : '0';
        
        document.body.appendChild(confetti);
        
        // Animación
        const animation = confetti.animate([
          {
            transform: 'translateY(0) rotate(0deg)',
            opacity: 1
          },
          {
            transform: `translateY(${window.innerHeight}px) rotate(${Math.random() * 720}deg)`,
            opacity: 0
          }
        ], {
          duration: 1500 + Math.random() * 2000,
          easing: 'cubic-bezier(0.2, 0.8, 0.3, 1)',
          delay: Math.random() * 500
        });
        
        // Eliminar después de la animación
        animation.onfinish = () => {
          confetti.remove();
        };
      }
    }

    // ====================
    // GESTIÓN DE GIROS
    // ====================
    async function handleSpin() {
      if (isSpinning) return;
      
      const totalBet = currentBet * activeLines;
      
      if (totalBet <= 0 || currentBet < CONFIG.MIN_BET || currentBet > CONFIG.MAX_BET) {
        setStatus("Configura una apuesta válida.");
        log("Apuesta inválida.", "system");
        return;
      }
      
      if (totalBet > balance && freeSpinsRemaining === 0) {
        setStatus("Saldo insuficiente.");
        log("Saldo insuficiente para la apuesta.", "system");
        return;
      }
      
      setSpinEnabled(false);
      
      // Cobrar apuesta (excepto en giros gratis)
      if (freeSpinsRemaining === 0) {
        balance -= totalBet;
        log(`Apuesta: ${formatCurrency(totalBet)} (${activeLines} líneas)`, "spin");
      } else {
        log(`Giro gratis ${CONFIG.FREE_SPINS_ON_SCATTER - freeSpinsRemaining + 1}/${CONFIG.FREE_SPINS_ON_SCATTER}`, "bonus");
      }
      
      // Animación de giro
      await animateSpin();
      
      // Generar resultado final
      fillGrid();
      
      // Evaluar resultados
      const { winningLines, totalWin, scatterCount } = evaluatePaylines();
      
      // VERIFICAR SI SE ACTIVA LA NORIA
      const wheelCheck = checkWheelActivation(winningLines, totalWin);
      
      // Actualizar saldo con ganancias normales
      if (totalWin > 0) {
        // Si se activa la noria, no añadimos la ganancia base aún
        if (wheelCheck.activate) {
          // Solo añadimos las ganancias que NO serán multiplicadas
          const winWithoutWheelLine = totalWin - wheelCheck.baseWin;
          if (winWithoutWheelLine > 0) {
            balance += winWithoutWheelLine;
          }
          lastWin = winWithoutWheelLine;
        } else {
          // Si no hay noria, añadir toda la ganancia
          balance += totalWin;
          lastWin = totalWin;
        }
      } else {
        lastWin = 0;
      }
      
      // Resaltar símbolos y líneas ganadoras
      highlightWinningSymbols(winningLines);
      highlightWinningLines(winningLines.map(wl => wl.lineIndex).filter(i => i >= 0));
      
      // Mostrar mensaje
      if (totalWin > 0) {
        const winLines = winningLines.filter(wl => wl.lineIndex >= 0).length;
        const hasScatter = winningLines.some(wl => wl.lineIndex === -1);
        
        let message = `¡Ganancia: ${formatCurrency(totalWin)}!`;
        if (winLines > 0) message += ` ${winLines} línea(s) ganadora(s).`;
        if (hasScatter) message += ` ${scatterCount} Scatters!`;
        
        setStatus(message);
        
        if (!wheelCheck.activate) {
          log(`Ganancia: ${formatCurrency(totalWin)}`, "win");
        }
      } else {
        setStatus("Sin premio en este giro.");
      }
      
      // Actualizar contadores
      winningLinesDisplay.textContent = winningLines.filter(wl => wl.lineIndex >= 0).length;
      
      // Manejar giros gratis
      if (freeSpinsRemaining > 0) {
        freeSpinsRemaining--;
        if (freeSpinsRemaining === 0) {
          log("Giros gratis completados.", "bonus");
        }
      }
      
      updateUI();
      
      // SI SE ACTIVÓ LA NORIA
      if (wheelCheck.activate) {
        // Pequeña pausa antes de mostrar la noria
        setTimeout(() => {
          activateWheel(wheelCheck.baseWin, wheelCheck.lineIndex);
        }, 1500);
      } else {
        // Continuar normal o con Auto Spin
        if (isAutoSpinning && autoSpinCount > 0) {
          autoSpinCount--;
          if (autoSpinCount > 0) {
            autoSpinButton.querySelector('span:last-child').textContent = `Auto (${autoSpinCount})`;
            setTimeout(() => {
              if (balance >= totalBet || freeSpinsRemaining > 0) {
                handleSpin();
              } else {
                stopAutoSpin();
                setSpinEnabled(true);
              }
            }, 1000);
          } else {
            stopAutoSpin();
            setSpinEnabled(true);
          }
        } else {
          setSpinEnabled(true);
        }
      }
    }

    // ====================
    // AUTO SPIN
    // ====================
    function startAutoSpin(count = CONFIG.AUTO_SPIN_COUNT) {
      if (isAutoSpinning || isSpinning) return;
      
      const totalBet = currentBet * activeLines;
      if (totalBet > balance && freeSpinsRemaining === 0) {
        setStatus("Saldo insuficiente para Auto Spin.");
        return;
      }
      
      isAutoSpinning = true;
      autoSpinCount = count;
      autoSpinButton.querySelector('span:last-child').textContent = `Auto (${autoSpinCount})`;
      autoSpinButton.style.background = "linear-gradient(120deg, #dc2626, #b91c1c)";
      
      log(`Auto Spin iniciado (${count} giros)`, "system");
      handleSpin();
    }

    function stopAutoSpin() {
      isAutoSpinning = false;
      autoSpinCount = 0;
      autoSpinButton.querySelector('span:last-child').textContent = "Auto (5)";
      autoSpinButton.style.background = "linear-gradient(120deg, #f59e0b, #d97706)";
    }

    // ====================
    // BONO AUTOMÁTICO
    // ====================
    function startBonusTimer() {
      setInterval(() => {
        countdown--;
        if (countdown <= 0) {
          countdown = CONFIG.COUNTDOWN_SECONDS;
          balance += CONFIG.BONUS_AMOUNT;
          updateUI();
          log(`Bono de fidelización: ${formatCurrency(CONFIG.BONUS_AMOUNT)}`, "bonus");
        }
        countdownElement.textContent = `Próximo bono: ${countdown} s`;
      }, 1000);
    }

    // ====================
    // EVENT LISTENERS
    // ====================
    function setupEventListeners() {
      // Botón SPIN
      spinButton.addEventListener("click", () => {
        if (!spinButton.disabled) {
          handleSpin();
        }
      });
      
      // Botón Auto Spin
      autoSpinButton.addEventListener("click", () => {
        if (isAutoSpinning) {
          stopAutoSpin();
          setSpinEnabled(true);
        } else {
          startAutoSpin();
        }
      });
      
      // Botón Max Bet
      maxBetButton.addEventListener("click", () => {
        currentBet = CONFIG.MAX_BET;
        activeLines = CONFIG.MAX_LINES;
        betInput.value = currentBet;
        linesSlider.value = activeLines;
        updateUI();
        drawPaylines();
        log(`Apuesta máxima configurada: ${formatCurrency(CONFIG.MAX_BET * CONFIG.MAX_LINES)}`, "system");
      });
      
      // Botón información
      infoButton.addEventListener("click", () => {
        paytableModal.classList.add("active");
      });
      
      closeModalBtn.addEventListener("click", () => {
        paytableModal.classList.remove("active");
      });
      
      // Cerrar modales al hacer clic fuera
      paytableModal.addEventListener("click", (e) => {
        if (e.target === paytableModal) {
          paytableModal.classList.remove("active");
        }
      });
      
      wheelModal.addEventListener("click", (e) => {
        if (e.target === wheelModal) {
          // No permitir cerrar durante la animación
          if (!spinWheelButton.disabled || spinWheelButton.textContent.includes("Recolectar")) {
            collectWheelWin();
          }
        }
      });
      
      // Control de apuesta
      betInput.addEventListener("change", () => {
        let newBet = parseInt(betInput.value) || CONFIG.MIN_BET;
        newBet = Math.max(CONFIG.MIN_BET, Math.min(CONFIG.MAX_BET, newBet));
        currentBet = newBet;
        betInput.value = currentBet;
        updateUI();
        
        betButtons.forEach(btn => {
          btn.classList.remove('active');
          if (parseInt(btn.dataset.bet) === currentBet) {
            btn.classList.add('active');
          }
        });
      });
      
      // Botones de apuesta rápida
      betButtons.forEach(btn => {
        btn.addEventListener('click', () => {
          currentBet = parseInt(btn.dataset.bet);
          betInput.value = currentBet;
          betButtons.forEach(b => b.classList.remove('active'));
          btn.classList.add('active');
          updateUI();
        });
      });
      
      // Control de líneas
      linesSlider.addEventListener("input", () => {
        activeLines = parseInt(linesSlider.value);
        updateUI();
        drawPaylines();
      });
      
      // Limpiar registro
      clearLogBtn.addEventListener("click", () => {
        logElement.innerHTML = "";
        log("Registro limpiado.", "system");
      });
      
      // Botón cerrar noria
      closeWheelModalBtn.addEventListener("click", () => {
        if (!spinWheelButton.disabled || spinWheelButton.textContent.includes("Recolectar")) {
          collectWheelWin();
        }
      });
      
      // Botón girar noria
      spinWheelButton.addEventListener("click", () => {
        if (spinWheelButton.textContent.includes("Recolectar")) {
          collectWheelWin();
        }
      });
      
      // Teclado shortcuts
      document.addEventListener("keydown", (e) => {
        if (e.code === "Space" && !spinButton.disabled) {
          e.preventDefault();
          handleSpin();
        }
        
        if (e.code === "KeyA" && e.ctrlKey) {
          e.preventDefault();
          if (isAutoSpinning) {
            stopAutoSpin();
          } else {
            startAutoSpin();
          }
        }
        
        if (e.code === "KeyM") {
          e.preventDefault();
          maxBetButton.click();
        }
        
        if (e.code === "Escape") {
          if (wheelModal.classList.contains("active")) {
            collectWheelWin();
          }
          if (paytableModal.classList.contains("active")) {
            paytableModal.classList.remove("active");
          }
        }
      });
    }

      startBonusTimer();
      setupEventListeners();
      createWheelSectors(); // Crear noria inicialmente
    }

    // Iniciar la aplicación
    init();
