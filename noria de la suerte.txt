    <script>
        // Wheel configuration (45 degrees per sector)
        const INDICATOR_ANGLE = 0; 

        const SECTORS = [
            // stopAngle is the center angle of the sector in the SVG coordinate system (0=right)
            { id: 'x50', multiplier: 50.0, probability: 0.05, stopAngle: 0 }, 
            { id: 'x25', multiplier: 25.0, probability: 0.05, stopAngle: 45 },  
            { id: 'x10', multiplier: 10.0, probability: 0.10, stopAngle: 90 },  
            { id: 'x5', multiplier: 5.0, probability: 0.10, stopAngle: 135 },   
            { id: 'x3', multiplier: 3.0, probability: 0.15, stopAngle: 180 },   
            { id: 'x2', multiplier: 2.0, probability: 0.15, stopAngle: 225 },   
            { id: 'x1_5', multiplier: 1.5, probability: 0.20, stopAngle: 270 }, 
            { id: 'x0', multiplier: 0.0, probability: 0.20, stopAngle: 315 },  
        ];

        // Game State
        let balance = 1000;
        let isSpinning = false;

        // DOM Elements
        const balanceDisplay = document.getElementById('balance-display');
        const betInput = document.getElementById('bet-input');
        const btnSpin = document.getElementById('btn-spin');
        const wheelSVG = document.getElementById('wheel-svg');
        const statusMessage = document.getElementById('status-message');
        const multiplierDisplay = document.getElementById('multiplier-display');
        const prizeDisplay = document.getElementById('prize-display');
        const logContainer = document.getElementById('log-container');

        // Utility function for currency formatting
        function formatCurrency(amount) {
            return amount.toLocaleString('es-ES', {
                style: 'currency',
                currency: 'EUR',
                minimumFractionDigits: 0,
                maximumFractionDigits: 0
            });
        }
        
        // --- Incremental Count Up Animation ---
        function animateCountUp(targetElement, finalAmount, duration = 2000) {
            const startAmount = 0;
            let currentAmount = startAmount;
            const stepTime = 10; // Update every 10ms
            const totalSteps = duration / stepTime;
            const stepValue = finalAmount / totalSteps;

            return new Promise(resolve => {
                const interval = setInterval(() => {
                    currentAmount += stepValue;

                    if (currentAmount >= finalAmount) {
                        clearInterval(interval);
                        currentAmount = finalAmount;
                        resolve();
                    }

                    // Format and display the current amount
                    targetElement.textContent = formatCurrency(Math.floor(currentAmount));
                }, stepTime);
            });
        }
        
        // Log events to history
        function log(message, type = 'info') {
            const entry = document.createElement('div');
            entry.className = 'log-entry py-1 ' + (type === 'win' ? 'text-green-400' : type === 'loss' ? 'text-red-400' : 'text-gray-500');
            entry.textContent = `[${new Date().toLocaleTimeString()}] ${message}`;
            logContainer.prepend(entry);
            if (logContainer.children.length > 20) {
                logContainer.lastChild.remove();
            }
        }

        // ----------------------------------------------------
        // LOGIC FOR SELECTION AND STOP ANGLE CALCULATION
        // ----------------------------------------------------

        /**
         * Selects a sector based on its probability (Weight)
         * @returns {object} The winning sector.
         */
        function selectWinningSector() {
            let totalWeight = SECTORS.reduce((sum, sector) => sum + sector.probability, 0);
            let randomNum = Math.random() * totalWeight;
            
            let cumulativeWeight = 0;
            for (const sector of SECTORS) {
                cumulativeWeight += sector.probability;
                if (randomNum < cumulativeWeight) {
                    return sector;
                }
            }
            // Fallback
            return SECTORS[SECTORS.length - 1]; 
        }

        /**
         * Calculates the exact angle where the wheel should stop.
         * @param {object} winningSector - The selected sector.
         * @returns {number} The total rotation angle.
         */
        function calculateStopAngle(winningSector) {
            const targetAngle = winningSector.stopAngle; 

            // Precision deviation: small variation within the sector (± 18 degrees).
            const deviation = (Math.random() * 36) - 18; // Range [-18, 18]

            // Rotation Offset (Indicator Angle is 0 degrees):
            let rotationOffset = 360 - ((targetAngle + deviation) % 360);
            
            // Ensure full spins (minimum 5) for visual effect.
            const fullSpins = 5;
            const fullRotation = fullSpins * 360;

            // Total Rotation Angle: Full spins + the offset to stop at the correct point.
            return fullRotation + rotationOffset;
        }

        // ----------------------------------------------------
        // GAME AND UI LOGIC
        // ----------------------------------------------------

        function updateUI() {
            balanceDisplay.textContent = formatCurrency(balance);
            // Disable button if spinning, insufficient balance, or bet is invalid
            btnSpin.disabled = isSpinning || balance < Number(betInput.value) || Number(betInput.value) <= 0;
        }

        /**
         * Clears all win/loss classes from result displays.
         */
        function clearResultClasses() {
            multiplierDisplay.classList.remove('win-text', 'loss-text', 'text-gray-500');
            prizeDisplay.classList.remove('win-text', 'loss-text');
        }

        async function spinWheel() {
            const bet = Number(betInput.value);

            if (isSpinning || bet <= 0 || balance < bet) return;

            // 1. Deduct bet and update UI
            isSpinning = true;
            balance -= bet;
            updateUI();
            
            // Clear messages and set spinning status
            clearResultClasses(); // Clear any previous coloring
            statusMessage.textContent = "¡GIRANDO RUEDA!";
            multiplierDisplay.textContent = "";
            prizeDisplay.textContent = "";

            // 2. Determine winner and stop angle
            const winningSector = selectWinningSector();
            const degrees = calculateStopAngle(winningSector);
            
            // 3. Apply rotation and CSS animation
            const spinDuration = 6000; // 6 seconds spin
            
            wheelSVG.style.transition = `transform ${spinDuration / 1000}s cubic-bezier(0.35, 0.01, 0, 1)`;
            wheelSVG.style.transform = `rotate(${degrees}deg)`;

            // Wait for the wheel to stop
            await new Promise(resolve => setTimeout(resolve, spinDuration + 50));

            // 4. Resolve the result
            isSpinning = false;
            
            // 4a. Reset transition and maintain final angle
            wheelSVG.style.transition = 'none'; 
            let finalRotation = degrees % 360;
            if (finalRotation < 0) finalRotation += 360; 
            wheelSVG.style.transform = `rotate(${finalRotation}deg)`;

            // 4b. Calculate Winnings
            const multiplier = winningSector.multiplier;
            const winAmount = bet * multiplier;

            if (multiplier > 0) {
                // WINNING CASE
                
                // Set text and apply win class
                multiplierDisplay.textContent = `¡GANAS! Multiplicador: ${multiplier.toFixed(2)}x`;
                multiplierDisplay.classList.add('win-text');
                
                // Animate the prize amount
                prizeDisplay.classList.add('win-text');
                statusMessage.textContent = "CALCULANDO PREMIO...";
                
                // Animation duration for counting up
                const countDuration = Math.min(2000, winAmount * 50); // Max 2 seconds, or 50ms per Euro

                await animateCountUp(prizeDisplay, winAmount, countDuration);
                
                balance += winAmount;

                // Final status message
                statusMessage.textContent = "¡PREMIO AÑADIDO AL SALDO!";
                log(`Apuesta ${formatCurrency(bet)}. Ganador ${multiplier.toFixed(2)}x. Premio: ${formatCurrency(winAmount)}. Saldo: ${formatCurrency(balance)}`, 'win');
                
            } else {
                // LOSING CASE (0x)
                
                // Set text and apply loss class
                multiplierDisplay.textContent = `¡FALLO! Multiplicador: 0x`;
                multiplierDisplay.classList.add('loss-text');
                
                prizeDisplay.textContent = formatCurrency(-bet);
                prizeDisplay.classList.add('loss-text');
                
                statusMessage.textContent = "PÉRDIDA TOTAL.";
                log(`Apuesta ${formatCurrency(bet)}. Resultado 0x. Pérdida: ${formatCurrency(bet)}. Saldo: ${formatCurrency(balance)}`, 'loss');
            }
            
            updateUI();
        }

        // ----------------------------------------------------
        // INITIALIZATION
        // ----------------------------------------------------

        function init() {
            // Ensure the wheel starts at 0 degrees.
            wheelSVG.style.transform = 'rotate(0deg)'; 

            balanceDisplay.textContent = formatCurrency(balance);
            
            // Event Listeners
            btnSpin.addEventListener('click', spinWheel);
            betInput.addEventListener('input', updateUI);
            
            // Initial UI update to enable the button if bet > 0
            updateUI();
            log("JUEGO INICIADO. Suerte.", "info");
        }

        window.onload = init;
    </script>
