    // LÓGICA DE PAGO

    function evaluatePaylines() {
      const winningLines = [];
      let totalWin = 0;
      let scatterCount = 0;
      
      // Contar scatters primero
      for (let r = 0; r < CONFIG.ROWS; r++) {
        for (let c = 0; c < CONFIG.COLS; c++) {
          if (SYMBOLS[grid[r][c].symbolIndex].type === "scatter") {
            scatterCount++;
          }
        }
      }
      
      // Evaluar líneas de pago
      for (let lineIndex = 0; lineIndex < activeLines; lineIndex++) {
        const line = PAYLINES[lineIndex];
        let firstSymbol = null;
        let consecutive = 0;
        let hasWild = false;
        
        for (let pos = 0; pos < line.length; pos++) {
          const [col, row] = line[pos];
          const symbol = SYMBOLS[grid[row][col].symbolIndex];
          
          if (symbol.type === "wild") {
            hasWild = true;
            consecutive++;
            if (firstSymbol === null && pos > 0) {
              // Wild puede continuar una línea
              continue;
            }
          } else if (firstSymbol === null || 
                    (symbol.type === "regular" && 
                     (firstSymbol.name === symbol.name || hasWild))) {
            if (firstSymbol === null) {
              firstSymbol = symbol;
            }
            consecutive++;
            hasWild = false; // Reset wild para próximo símbolo
          } else {
            break;
          }
        }
        
        // Verificar si hay combinación ganadora (3+ símbolos)
        if (consecutive >= 3 && firstSymbol && firstSymbol.type === "regular") {
          const multiplier = firstSymbol.multiplier[consecutive - 1] || 0;
          if (multiplier > 0) {
            const win = currentBet * multiplier;
            totalWin += win;
            winningLines.push({
              lineIndex,
              symbol: firstSymbol.name,
              count: consecutive,
              multiplier,
              win
            });
          }
        }
      }
      
      // Pago de scatters
      if (scatterCount >= 3) {
        const scatterSymbol = SYMBOLS.find(s => s.type === "scatter");
        const multiplier = scatterSymbol.multiplier[scatterCount - 1] || 0;
        if (multiplier > 0) {
          const scatterWin = (currentBet * activeLines) * multiplier;
          totalWin += scatterWin;
          
          winningLines.push({
            lineIndex: -1, // -1 indica scatter
            symbol: "Scatter",
            count: scatterCount,
            multiplier,
            win: scatterWin
          });
        }
        
        // Activar giros gratis si es la primera vez
        if (freeSpinsRemaining === 0) {
          freeSpinsRemaining = CONFIG.FREE_SPINS_ON_SCATTER;
          log(`¡${scatterCount} Scatters! ${freeSpinsRemaining} giros gratis activados.`, "bonus");
        }
      }
      
      return { winningLines, totalWin, scatterCount };
    }

    async function handleSpin() {
      if (isSpinning) return;
      
      const totalBet = currentBet * activeLines;
      
      if (totalBet <= 0 || currentBet < CONFIG.MIN_BET || currentBet > CONFIG.MAX_BET) {
        setStatus("Configura una apuesta válida.");
        log("Apuesta inválida.", "system");
        return;
      }
      
      if (totalBet > balance) {
        setStatus("Saldo insuficiente.");
        log("Saldo insuficiente para la apuesta.", "system");
        return;
      }
      
      setSpinEnabled(false);
      
      // Cobrar apuesta (excepto en giros gratis)
      if (freeSpinsRemaining === 0) {
        balance -= totalBet;
        log(`Apuesta: ${formatCurrency(totalBet)} (${activeLines} líneas)`, "spin");
      } else {
        log(`Giro gratis ${CONFIG.FREE_SPINS_ON_SCATTER - freeSpinsRemaining + 1}/${CONFIG.FREE_SPINS_ON_SCATTER}`, "bonus");
      }
      
      // Animación de giro
      await animateSpin();
      
      // Generar resultado final
      fillGrid();
      
      // Evaluar resultados
      const { winningLines, totalWin, scatterCount } = evaluatePaylines();
      
      // Actualizar saldo y mostrar resultados
      if (totalWin > 0) {
        balance += totalWin;
        lastWin = totalWin;
        
        // Resaltar símbolos y líneas ganadoras
        highlightWinningSymbols(winningLines);
        highlightWinningLines(winningLines.map(wl => wl.lineIndex).filter(i => i >= 0));
        
        // Mostrar mensaje
        const winLines = winningLines.filter(wl => wl.lineIndex >= 0).length;
        const hasScatter = winningLines.some(wl => wl.lineIndex === -1);
        
        let message = `¡Ganancia: ${formatCurrency(totalWin)}!`;
        if (winLines > 0) message += ` ${winLines} línea(s) ganadora(s).`;
        if (hasScatter) message += ` ${scatterCount} Scatters!`;
        
        setStatus(message);
        log(`Ganancia: ${formatCurrency(totalWin)}`, "win");
        
        // Sonido de ganancia (simulado)
        playWinSound(totalWin);
      } else {
        lastWin = 0;
        setStatus("Sin premio en este giro.");
      }
      
      // Actualizar contadores
      winningLinesDisplay.textContent = winningLines.filter(wl => wl.lineIndex >= 0).length;
      
      // Manejar giros gratis
      if (freeSpinsRemaining > 0) {
        freeSpinsRemaining--;
        if (freeSpinsRemaining === 0) {
          log("Giros gratis completados.", "bonus");
        }
      }
      
      updateUI();
      
      // Continuar Auto Spin si está activo
      if (isAutoSpinning && autoSpinCount > 0) {
        autoSpinCount--;
        if (autoSpinCount > 0) {
          autoSpinButton.querySelector('span:last-child').textContent = `Auto (${autoSpinCount})`;
          setTimeout(() => {
            if (balance >= totalBet) {
              handleSpin();
            } else {
              stopAutoSpin();
              setSpinEnabled(true);
            }
          }, 1000);
        } else {
          stopAutoSpin();
          setSpinEnabled(true);
        }
      } else {
        setSpinEnabled(true);
      }
    }

        }
        countdownElement.textContent = `Próximo bono: ${countdown} s`;
      }, 1000);
    }

    // ====================
    // EVENT LISTENERS
    // ====================
    function setupEventListeners() {
      // Botón SPIN
      spinButton.addEventListener("click", () => {
        if (!spinButton.disabled) {
          handleSpin();
        }
      });
      
      // Botón Auto Spin
      autoSpinButton.addEventListener("click", () => {
        if (isAutoSpinning) {
          stopAutoSpin();
          setSpinEnabled(true);
        } else {
          startAutoSpin();
        }
      });
      
      // Botón Max Bet
      maxBetButton.addEventListener("click", () => {
        currentBet = CONFIG.MAX_BET;
        activeLines = CONFIG.MAX_LINES;
        betInput.value = currentBet;
        linesSlider.value = activeLines;
        updateUI();
        drawPaylines();
        log(`Apuesta máxima configurada: ${formatCurrency(CONFIG.MAX_BET * CONFIG.MAX_LINES)}`, "system");
      });
      
      // Botón información (tabla de pagos)
      infoButton.addEventListener("click", () => {
        paytableModal.classList.add("active");
      });
      
      closeModalBtn.addEventListener("click", () => {
        paytableModal.classList.remove("active");
      });
      
      // Cerrar modal al hacer clic fuera
      paytableModal.addEventListener("click", (e) => {
        if (e.target === paytableModal) {
          paytableModal.classList.remove("active");
        }
      });
      
      // Control de apuesta
      betInput.addEventListener("change", () => {
        let newBet = parseInt(betInput.value) || CONFIG.MIN_BET;
        newBet = Math.max(CONFIG.MIN_BET, Math.min(CONFIG.MAX_BET, newBet));
        currentBet = newBet;
        betInput.value = currentBet;
        updateUI();
        
        // Actualizar botones de apuesta
        betButtons.forEach(btn => {
          btn.classList.remove('active');
          if (parseInt(btn.dataset.bet) === currentBet) {
            btn.classList.add('active');
          }
        });
      });
      
      // Botones de apuesta rápida
      betButtons.forEach(btn => {
        btn.addEventListener('click', () => {
          currentBet = parseInt(btn.dataset.bet);
          betInput.value = currentBet;
          betButtons.forEach(b => b.classList.remove('active'));
          btn.classList.add('active');
          updateUI();
        });
      });
      
      // Control de líneas
      linesSlider.addEventListener("input", () => {
        activeLines = parseInt(linesSlider.value);
        updateUI();
        drawPaylines();
      });
      
      // Limpiar registro
      clearLogBtn.addEventListener("click", () => {
        logElement.innerHTML = "";
        log("Registro limpiado.", "system");
      });
      
      // Teclado shortcuts
      document.addEventListener("keydown", (e) => {
        if (e.code === "Space" && !spinButton.disabled) {
          e.preventDefault();
          handleSpin();
        }
        
        if (e.code === "KeyA" && e.ctrlKey) {
          e.preventDefault();
          if (isAutoSpinning) {
            stopAutoSpin();
          } else {
            startAutoSpin();
          }
        }
        
        if (e.code === "KeyM") {
          e.preventDefault();
          maxBetButton.click();
        }
      });
    }

    // ====================
    // INICIALIZACIÓN
    // ====================
    function init() {
      createGrid();
      fillGrid();
      updateUI();
      setSpinEnabled(true);
      setStatus("¡Bienvenido a Le Bandit Style! Configura tu apuesta y pulsa SPIN.");
      log("Sesión iniciada. ¡Buena suerte!", "system");
      startBonusTimer();
      setupEventListeners();
    }

    // Iniciar la aplicación
    init();
  </script>
