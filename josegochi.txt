        // Importaciones de Firebase

        // Variables globales que deben ser definidas por el entorno Canvas

        let db;

        let userId;
        let isAuthReady = false;

        // Estado del Pixelito (Valores iniciales)
        let petState = {
            bornTime: Date.now(), // Marca de tiempo de nacimiento
            lastUpdate: Date.now(), // Última actualización
            lifeSeconds: 0, // Segundos totales de vida simulada
            health: 100, // Salud (HP)
            hunger: 0, // Hambre (Decae la salud si es alto)
            happiness: 100, // Felicidad (Decae la salud si es bajo)
            stage: 1, // Etapa de evolución
            // La variable timeScale fue eliminada por petición del usuario.
        };
        
        // --- Configuraciones ---

        // Configuración de la evolución
        const evolutionStages = {
            1: { name: "Huevo", minDays: 0, pixel: "( o )", color: "text-white" },
            2: { name: "Bebé", minDays: 1, pixel: "ʕ •ᴥ•ʔ", color: "text-yellow-400" },
            3: { name: "Adolescente", minDays: 5, pixel: "( ಠ_ಠ )", color: "text-purple-400" },
            4: { name: "Adulto", minDays: 15, pixel: "(*_*)", color: "text-green-400" },
        };
        
        // Configuración de las frases/logros (días de vida)
        // Estas frases se muestran en el panel azul superior.
        const phrasesConfig = [
            { days: 0.05, text: "¡Hola Mundo! Nací hace poco." },
            { days: 1, text: "¡Sobreviví un día completo! 💪" },
            { days: 5, text: "¡Cinco días y sigo fuerte! 🌱" },

        // --- Utilidades ---

        /**
         * Muestra un mensaje temporal en el área de acción.
         * @param {string} message Mensaje a mostrar.
         */
        function displayActionMessage(message) {
            actionMessageEl.textContent = message;
            actionMessageEl.classList.remove('opacity-0');
            setTimeout(() => {
                actionMessageEl.classList.add('opacity-0');
            }, 3000);
        }

        /**
         * Convierte segundos en una cadena legible de tiempo completo.
         * @param {number} totalSeconds Segundos totales.
         * @returns {string} Cadena formateada (Y, M, D, h, m, s).
         */
        function formatLifeTime(totalSeconds) {
            if (totalSeconds < 0) totalSeconds = 0;
            
            // Constantes de tiempo
            const SECONDS_IN_MINUTE = 60;
            const SECONDS_IN_HOUR = 3600;
            const SECONDS_IN_DAY = 86400;
            // Usamos aproximaciones simples para meses/años para mantener la precisión
            const DAYS_IN_YEAR = 365.25; 
            const DAYS_IN_MONTH = 30.44; 
            
            const SECONDS_IN_MONTH = DAYS_IN_MONTH * SECONDS_IN_DAY; 
            const SECONDS_IN_YEAR = DAYS_IN_YEAR * SECONDS_IN_DAY; 

            let remainingSeconds = Math.floor(totalSeconds);

            const years = Math.floor(remainingSeconds / SECONDS_IN_YEAR);
            remainingSeconds %= SECONDS_IN_YEAR;

            const months = Math.floor(remainingSeconds / SECONDS_IN_MONTH);
            remainingSeconds %= SECONDS_IN_MONTH;

            const days = Math.floor(remainingSeconds / SECONDS_IN_DAY);
            remainingSeconds %= SECONDS_IN_DAY;

            const hours = Math.floor(remainingSeconds / SECONDS_IN_HOUR);
            remainingSeconds %= SECONDS_IN_HOUR;

            const minutes = Math.floor(remainingSeconds / SECONDS_IN_MINUTE);
            remainingSeconds %= SECONDS_IN_MINUTE;

            const seconds = remainingSeconds;

            const parts = [];
            if (years > 0) parts.push(`${years}a`);
            if (months > 0) parts.push(`${months}m`);
            if (days > 0) parts.push(`${days}d`);
            if (hours > 0) parts.push(`${hours}h`);
            if (minutes > 0) parts.push(`${minutes}min`);
            parts.push(`${seconds}s`); // Siempre mostrar segundos

            return parts.join(' ');
        }

        /**
         * Aplica los valores del estado a la interfaz.
         */
        function renderPet() {
            const { lifeSeconds, health, hunger, happiness, stage } = petState;

            // 1. Actualizar el contador de vida
            lifeDisplayEl.textContent = formatLifeTime(lifeSeconds);

            // 2. Actualizar barras de estado
            // En el boceto, las barras tienen un color sólido simple
            healthEl.style.width = `${health}%`;
            healthEl.className = `h-full rounded transition-all duration-500 bg-red-500`;
            
            hungerEl.style.width = `${hunger}%`;
            hungerEl.className = `h-full rounded transition-all duration-500 bg-yellow-500`;

            happinessEl.style.width = `${happiness}%`;
            happinessEl.className = `h-full rounded transition-all duration-500 bg-green-500`;

            // 3. Actualizar la apariencia del Pixelito
            const currentStage = evolutionStages[stage] || evolutionStages[1];
            pixelitoEl.textContent = currentStage.pixel;
            pixelitoEl.className = `pixel-font text-5xl transition-colors duration-500 ${currentStage.color}`;
        }
        
        // --- Lógica del Juego ---

        /**
         * Reinicia el estado del Pixelito.
         */
        function resetPet() {
            // Usamos un modal de confirmación simple, ya que alert() no debe usarse
            const modal = document.getElementById('reset-modal');
            modal.classList.remove('hidden');

            document.getElementById('confirm-reset').onclick = async () => {
                modal.classList.add('hidden');
                const now = Date.now();
                petState = {
                    bornTime: now,
                    lastUpdate: now,
                    lifeSeconds: 0,
                    health: 100,
                    hunger: 0,
                    happiness: 100,
                    stage: 1,
                };
                checkPhrases(); // Establece la frase inicial
                
                await savePetState();
                renderPet();
                displayActionMessage("¡Pixelito Reiniciado! (Etapa Huevo)");
                startGameLoop(); // Asegurar que el loop empiece si estaba muerto
            };

            document.getElementById('cancel-reset').onclick = () => {
                modal.classList.add('hidden');
            };
        }

        /**
         * Aplica la acción a las estadísticas.
         * @param {string} type Tipo de acción ('eat', 'play', 'sleep', 'study', 'gym', 'clean', 'work').
         */
        function applyAction(type) {
            let message = '';
            
            // Los valores están diseñados para ser 'generosos' para compensar el rápido deterioro simulado.
            switch (type) {
                case 'eat': // Alimentar
                    petState.hunger = Math.max(0, petState.hunger - 40);
                    petState.health = Math.min(100, petState.health + 10);
                    message = "¡Ñam ñam! Pixelito está comiendo.";
                    break;
                case 'study': // Estudio
                    petState.happiness = Math.min(100, petState.happiness + 20);
                    petState.hunger = Math.min(100, petState.hunger + 20); // El esfuerzo mental da hambre
                    petState.health = Math.min(100, petState.health + 5); // Un poco de salud por la concentración
                    message = "Pixelito está estudiando. ¡Qué aplicado!";
                    break;
                case 'sleep': // Dormir
                    petState.health = Math.min(100, petState.health + 30);
                    petState.happiness = Math.min(100, petState.happiness + 10);
                    message = "Zzz... Pixelito está durmiendo profundamente.";
                    break;
                case 'play': // Jugar
                    petState.happiness = Math.min(100, petState.happiness + 30);
                    petState.hunger = Math.min(100, petState.hunger + 10); // Jugar da hambre
                    message = "¡Wiiii! Pixelito está jugando a la pelota.";
                    break;
                case 'gym': // Gimnasio
                    petState.health = Math.min(100, petState.health + 20);
                    petState.hunger = Math.min(100, petState.hunger + 20); // El ejercicio da hambre
                    petState.happiness = Math.max(0, petState.happiness - 5); // Un poco de esfuerzo
                    message = "¡Fuerza! Pixelito está en el gimnasio.";
                    break;
                case 'evolution': // Evo (llamada directa, pero la dejo por si cambia el nombre del botón)
                    checkEvolution();
                    return; // Ya tiene su propio mensaje
                case 'clean': // Limpiar
                    petState.happiness = Math.min(100, petState.happiness + 30);
                    petState.health = Math.min(100, petState.health + 10);
                    message = "Pixelito está limpio y feliz. ✨";
                    break;
                case 'work': // Trabajar
                    petState.hunger = Math.min(100, petState.hunger + 30); // El trabajo es estresante y quema energía
                    petState.happiness = Math.max(0, petState.happiness - 10); // El trabajo no siempre es divertido
                    petState.health = Math.min(100, petState.health + 10); // Recompensa base
                    message = "Pixelito está trabajando duro por un futuro mejor. 💼";
                    break;
                case 'reset': // Reset (llamada directa, pero la dejo por si cambia el nombre del botón)
                    resetPet();
                    return; // Ya tiene su propio mensaje
            }
            
            savePetState();
            renderPet();
            displayActionMessage(message);
        }
        
        /**
         * Verifica si Pixelito debe evolucionar y lo actualiza si es necesario.
         */
        function checkEvolution() {
            const currentStage = petState.stage;
            const currentDays = petState.lifeSeconds / 86400; // Segundos por día
            let highestStageReached = currentStage;
            
            // Determinar la etapa más alta que ha alcanzado
            for (const key in evolutionStages) {
                const stageData = evolutionStages[key];
                if (currentDays >= stageData.minDays) {
                    highestStageReached = parseInt(key);
                }
            }

            if (highestStageReached > currentStage) {
                // Evoluciona a la etapa más alta alcanzada
                petState.stage = highestStageReached;
                savePetState();
                renderPet();
                checkPhrases(); // Actualizar la frase tras la evolución
                displayActionMessage(`¡EVOLUCIÓN! Pixelito ahora es ${evolutionStages[highestStageReached].name}.`);
                return true;
            } else {
                 displayActionMessage("Aún no tienes días suficientes para evolucionar.");
            }
            return false;
        }
        
        /**
         * Verifica y muestra la frase/logro más reciente.
         */
        function checkPhrases() {
            const currentDays = petState.lifeSeconds / 86400;
            let achievedPhrase = "Pixelito está tranquilo."; // Mensaje por defecto

            // Buscar la frase más reciente que se haya desbloqueado
            for (const phrase of phrasesConfig) {
                if (currentDays >= phrase.days) {
                    achievedPhrase = phrase.text;
                }
            }

            phraseEl.textContent = achievedPhrase;
            // No mostrar mensaje de acción ya que esto se llama automáticamente al inicio y evolución.
        }

        // --- Bucle principal del juego (Game Loop) ---
        let gameLoopInterval;
        let lastTime = Date.now();
        const MS_PER_SEC = 1000;

        /**
         * Bucle de actualización del juego (cada 1 segundo de tiempo real).
         */
        function gameLoop() {
            const now = Date.now();
            // Tiempo real transcurrido desde la última actualización (en segundos)
            const dtReal = (now - lastTime) / MS_PER_SEC; 
            lastTime = now;

            // Tiempo simulado transcurrido (sin timeScale, usa dtReal directamente)
            const dtSimulated = dtReal; 

            // 1. Actualizar el tiempo de vida
            petState.lifeSeconds += dtSimulated;

            // 2. Aplicar el deterioro de las estadísticas (basado en el tiempo simulado)
            // Velocidad de deterioro base
            const decayFactor = dtSimulated * 0.05; 

            petState.hunger = Math.min(100, petState.hunger + decayFactor); // El hambre aumenta
            petState.happiness = Math.max(0, petState.happiness - decayFactor * 0.5); // La felicidad decae
            
            // Deterioro de salud
            let healthDecay = decayFactor * 0.2;
            if (petState.hunger >= 80) healthDecay += decayFactor * 0.5; // Mucha hambre = más daño
            if (petState.happiness <= 20) healthDecay += decayFactor * 0.5; // Poca felicidad = más daño
            petState.health = Math.max(0, petState.health - healthDecay);

            // 3. Comprobar si Pixelito muere
            if (petState.health <= 0) {
                clearInterval(gameLoopInterval);
                pixelitoEl.textContent = "X_X";
                displayActionMessage("¡Pixelito ha muerto! Reinicia para empezar de nuevo.");
                savePetState(); 
                return; 
            }

            // 4. Comprobar evolución silenciosamente (para que el estado se actualice automáticamente)
            const currentDays = petState.lifeSeconds / 86400;
            let highestStageReached = petState.stage;
            for (const key in evolutionStages) {
                if (currentDays >= evolutionStages[key].minDays) {
                    highestStageReached = parseInt(key);
                }
            }
            if (highestStageReached > petState.stage) {
                petState.stage = highestStageReached;
            }

            // 5. Guardar el estado periódicamente y renderizar
            if (Math.floor(petState.lifeSeconds) % 10 === 0) {
                petState.lastUpdate = now;
                savePetState();
            }

            renderPet();
        }

        /**
         * Inicia o reanuda el bucle del juego.
         */
        function startGameLoop() {
             if (gameLoopInterval) clearInterval(gameLoopInterval);
             lastTime = Date.now(); 
             renderPet();
             // El bucle se ejecuta cada 100ms de tiempo real (10 veces por segundo)
             gameLoopInterval = setInterval(gameLoop, 100); 
        }

        // --- Lógica de persistencia (Firestore) ---
        
        /**
         * Guarda el estado actual del Pixelito en Firestore.
         */
        async function savePetState() {
            if (!isAuthReady || !userId) {
                console.warn("Auth no lista o userId no disponible. Guardando en memoria.");
                return;
            }

            try {
                // Ruta para el estado privado del usuario: /artifacts/{appId}/users/{userId}/pet/state
                const docRef = doc(db, 'artifacts', appId, 'users', userId, 'pet', 'state');
                await setDoc(docRef, petState);
            } catch (e) {
                console.error("Error al guardar en Firestore:", e);
                displayActionMessage("Error al guardar el estado.");
            }
        }

        /**
         * Inicializa Firebase, autentica y carga el estado del Pixelito.
         */
        async function initializeFirebaseAndLoad() {
            try {
                // 1. Inicializar Firebase
                app = initializeApp(firebaseConfig);
                db = getFirestore(app);
                auth = getAuth(app);
                setLogLevel('error'); 

                // 2. Autenticación (o usar token si está disponible)
                if (initialAuthToken) {
                    await signInWithCustomToken(auth, initialAuthToken);
                } else {
                    await signInAnonymously(auth);
                }

                // 3. Listener de estado de autenticación
                onAuthStateChanged(auth, async (user) => {
                    if (user) {
                        userId = user.uid;
                        userIdDisplayEl.textContent = userId.substring(0, 8) + '...';
                        isAuthReady = true;

                        // 4. Listener del documento de Firestore
                        const docRef = doc(db, 'artifacts', appId, 'users', userId, 'pet', 'state');
                        
                        onSnapshot(docRef, (docSnapshot) => {
                            if (docSnapshot.exists()) {
                                const savedState = docSnapshot.data();
                                
                                // Si el Pixelito está "muerto", no actualizamos el tiempo.
                                if (savedState.health <= 0) {
                                    petState = savedState;
                                    renderPet();
                                    clearInterval(gameLoopInterval);
                                    pixelitoEl.textContent = "X_X";
                                    displayActionMessage("Pixelito está muerto. ¡Reinicia!");
                                    return;
                                }

                                // 4.1 Cargar y reconciliar el estado
                                petState = savedState;
                                
                                // 4.2 Reconciliar el tiempo de vida simulado (para la persistencia)
                                const now = Date.now();
                                const timePassedReal = now - petState.lastUpdate; 
                                // Ya no se usa timeScale
                                const timeToAddSimulated = (timePassedReal / 1000); 

                                petState.lifeSeconds += timeToAddSimulated;
                                petState.lastUpdate = now; 
                                
                                // 4.3 Comenzar el bucle de juego con el estado actualizado
                                startGameLoop();
                                checkPhrases(); // Establece la frase inicial al cargar

                            } else {
                                // Si no hay datos, guardar el estado inicial y comenzar el juego
                                savePetState();
                                startGameLoop();
                                checkPhrases(); // Establece la frase inicial
                            }
                        }, (error) => {
                            console.error("Error en onSnapshot de Firestore:", error);
                            displayActionMessage("Error al conectar con la base de datos.");
                        });

                    } else {
                        isAuthReady = true; 
                        userIdDisplayEl.textContent = 'Temporal';
                        startGameLoop();
                        checkPhrases(); // Establece la frase inicial
                    }
                });

            } catch (e) {
                console.error("Error al inicializar Firebase:", e);
                displayActionMessage("Error grave al iniciar la aplicación.");
            }
        }

        // Cargar iconos y empezar la inicialización
        document.addEventListener('DOMContentLoaded', () => {
            // Fila 1
            document.getElementById('btn-eat').onclick = () => applyAction('eat');
            document.getElementById('btn-study').onclick = () => applyAction('study');
            document.getElementById('btn-sleep').onclick = () => applyAction('sleep');
            
            // Fila 2
            document.getElementById('btn-play').onclick = () => applyAction('play');
            document.getElementById('btn-gym').onclick = () => applyAction('gym');
            document.getElementById('btn-evolution').onclick = checkEvolution;
            
            // Fila 3
            document.getElementById('btn-clean').onclick = () => applyAction('clean');
            document.getElementById('btn-work').onclick = () => applyAction('work');
            document.getElementById('btn-reset').onclick = resetPet;

            lucide.createIcons();
            initializeFirebaseAndLoad();
        });
